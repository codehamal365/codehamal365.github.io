## 什么是Electron？

### 定义
Electron 是一个使用 JavaScript、HTML 和 CSS 构建跨平台桌面应用程序的框架。它基于 Node.js 和 Chromium，允许你使用 Web 技术开发桌面应用。

### 核心特点
- **跨平台**: 支持 Windows、macOS、Linux
- **Web技术**: 使用熟悉的 HTML、CSS、JavaScript
- **Node.js集成**: 可以访问操作系统 API
- **Chromium内核**: 现代化的渲染引擎
- **丰富的生态系统**: 大量的第三方库和工具

### 应用场景
- 桌面办公软件
- 开发者工具
- 媒体播放器
- 通讯软件
- 数据可视化工具

## Electron架构

### 进程模型

```
{% raw %}
┌─────────────────────────────────────────────────────────────┐
│                     Electron 应用架构                         │
├─────────────────────────────────────────────────────────────┤
│                                                              │
│  ┌─────────────────────────────────────────────────────┐    │
│  │                 主进程 (Main Process)                │    │
│  │  • 运行在 Node.js 环境中                            │    │
│  │  • 可以访问所有 Node.js API                         │    │
│  │  • 负责应用生命周期管理                             │    │
│  │  • 创建和管理窗口                                   │    │
│  │  • 处理系统事件                                     │    │
│  │  • 访问操作系统 API                                 │    │
│  └─────────────────────────────────────────────────────┘    │
│                           │                                   │
│                           │ IPC 通信                           │
│                           ▼                                   │
│  ┌─────────────────────────────────────────────────────┐    │
│  │              预加载脚本 (Preload Script)             │    │
│  │  • 运行在渲染进程和主进程之间                        │    │
│  │  • 使用 contextBridge 暴露 API                       │    │
│  │  • 提供类型安全的接口                                │    │
│  │  • 隔离渲染进程和主进程                              │    │
│  └─────────────────────────────────────────────────────┘    │
│                           │                                   │
│                           │ contextBridge                     │
│                           ▼                                   │
│  ┌─────────────────────────────────────────────────────┐    │
│  │            渲染进程 (Renderer Process)               │    │
│  │  • 运行在浏览器环境中                                │    │
│  │  • 不能直接访问 Node.js API                          │    │
│  │  • 负责 UI 渲染                                      │    │
│  │  • 通过预加载脚本与主进程通信                        │    │
│  └─────────────────────────────────────────────────────┘    │
│                                                              │
└─────────────────────────────────────────────────────────────┘
{% endraw %}
```

### 进程对比

| 特性 | 主进程 | 渲染进程 |
|------|--------|----------|
| 运行环境 | Node.js | 浏览器 |
| Node.js API | ✅ 完全访问 | ❌ 受限访问 |
| DOM 访问 | ❌ 无 | ✅ 完全访问 |
| 窗口管理 | ✅ 可以创建 | ❌ 不能创建 |
| 系统 API | ✅ 可以访问 | ❌ 受限访问 |
| IPC 通信 | ✅ 发送/接收 | ✅ 发送/接收 |

## 快速开始

### 最小示例

#### 1. 修改 package.json
```json
{% raw %}
{
  "name": "my-electron-app",
  "version": "1.0.0",
  "main": "main.js",
  "scripts": {
    "start": "electron ."
  },
  "dependencies": {
    "electron": "^40.0.0"
  }
}
{% endraw %}
```

#### 2. 创建主进程 (main.js)
```javascript
{% raw %}
const { app, BrowserWindow } = require('electron');

// 创建主窗口
function createWindow() {
  const mainWindow = new BrowserWindow({
    width: 800,
    height: 600,
    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,
    },
  });

  // 加载 HTML 文件
  mainWindow.loadFile('index.html');
}

// 应用启动
app.whenReady().then(() => {
  createWindow();

  // macOS 特殊处理
  app.on('activate', () => {
    if (BrowserWindow.getAllWindows().length === 0) {
      createWindow();
    }
  });
});

// 应用关闭
app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') {
    app.quit();
  }
});
{% endraw %}
```

```html
    <p>这是一个使用 Web 技术构建的桌面应用。</p>
</body>
</html>
{% endraw %}
```

#### 4. 运行应用

## IPC通信

### 预加载脚本 (preload.js)
```javascript
{% raw %}
const { contextBridge, ipcRenderer } = require('electron');

// 安全地暴露 API 给渲染进程
contextBridge.exposeInMainWorld('electronAPI', {
  // 发送消息到主进程
  send: (channel, data) => {
    ipcRenderer.send(channel, data);
  },

  // 接收主进程消息
  receive: (channel, callback) => {
    ipcRenderer.on(channel, (event, ...args) => callback(...args));
  },

  // 调用主进程方法 (异步)
  invoke: (channel, data) => {
    return ipcRenderer.invoke(channel, data);
  },

  // 移除监听器
  removeListener: (channel, callback) => {
    ipcRenderer.removeListener(channel, callback);
  },
});
{% endraw %}
```

### 主进程 IPC 处理
```javascript
{% raw %}
const { ipcMain, dialog } = require('electron');

// 处理同步消息
ipcMain.on('synchronous-message', (event, arg) => {
  console.log('收到同步消息:', arg);
  event.returnValue = '同步回复';
});

// 处理异步消息
ipcMain.on('asynchronous-message', (event, arg) => {
  console.log('收到异步消息:', arg);
  event.reply('asynchronous-reply', '异步回复');
});

// 处理异步方法调用
ipcMain.handle('show-open-dialog', async (event, options) => {
  const result = await dialog.showOpenDialog(options);
  return result;
});

// 处理带参数的调用
ipcMain.handle('get-file-contents', async (event, filePath) => {
  const fs = require('fs').promises;
  try {
    const contents = await fs.readFile(filePath, 'utf8');
    return { success: true, contents };
  } catch (error) {
    return { success: false, error: error.message };
  }
});
{% endraw %}
```

### 渲染进程调用
```javascript
{% raw %}
// 同步通信
const syncResult = window.electronAPI.send('synchronous-message', 'Hello');
console.log('同步结果:', syncResult);

// 异步通信
window.electronAPI.send('asynchronous-message', 'Hello');
window.electronAPI.receive('asynchronous-reply', (data) => {
  console.log('异步回复:', data);
});

// 调用异步方法
async function openFile() {
  try {
    const result = await window.electronAPI.invoke('show-open-dialog', {
      title: '选择文件',
      properties: ['openFile'],
    });

    if (!result.canceled) {
      console.log('选择的文件:', result.filePaths[0]);

      // 读取文件内容
      const fileResult = await window.electronAPI.invoke('get-file-contents', result.filePaths[0]);
      if (fileResult.success) {
        console.log('文件内容:', fileResult.contents);
      }
    }
  } catch (error) {
    console.error('操作失败:', error);
  }
}
{% endraw %}
```

```javascript
{% raw %}

// 渲染进程

```

#### 2. 事件通知模式
```javascript
{% raw %}
// 主进程
ipcMain.on('subscribe-updates', (event) => {
  // 订阅数据变化
  const subscription = dataStore.subscribe((data) => {
    event.reply('data-updated', data);
  });

  // 清理订阅
  event.on('close', () => {
    subscription.unsubscribe();
  });
});

// 渲染进程
window.electronAPI.send('subscribe-updates');
window.electronAPI.receive('data-updated', (data) => {
  console.log('数据更新:', data);
});
{% endraw %}
```

#### 3. 双向通信模式
```javascript
{% raw %}
// 主进程
ipcMain.on('progress-update', (event, progress) => {
  // 处理进度更新
  console.log('进度:', progress);

  // 发送确认
  event.reply('progress-ack', { received: true });
});

// 渲染进程
window.electronAPI.send('progress-update', 50);
window.electronAPI.receive('progress-ack', (ack) => {
  console.log('确认收到:', ack);
});
{% endraw %}
```

### 使用 electron-builder

#### 1. 安装 electron-builder
```bash
{% raw %}
npm install electron-builder --save-dev

```

#### 2. 配置 package.json
```json
{% raw %}
{
  "build": {
    "appId": "com.example.myapp",
    "productName": "我的应用",
    "directories": {
      "output": "release"

      "dist/**/*",
      "node_modules/**/*",

        "nsis",
        "portable"

      "publisherName": "Example Inc."
    },
    "linux": {
      "target": [
        "AppImage",
        "deb"
      ],
```

#### 3. 准备图标
- **macOS**: `build/icon.icns` (512x512)
- **Windows**: `build/icon.ico` (256x256)
- **Linux**: `build/icon.png` (512x512)

#### 4. 构建应用
```bash
{% raw %}

npm run dist

# 仅构建 macOS
npm run dist:mac
```
