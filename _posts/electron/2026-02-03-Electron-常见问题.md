## 安装和配置问题

### 1. Electron 安装失败

**问题描述**：
```bash
{% raw %}
npm install electron --save-dev
# 安装失败或超时
{% endraw %}
```

**可能原因**：
- 网络问题
- npm 镜像源问题
- 系统权限问题
- Node.js 版本不兼容

**解决方案**：

#### 方案1：使用淘宝镜像
```bash
{% raw %}
# 设置 Electron 镜像
npm config set electron_mirror https://npmmirror.com/mirrors/electron/

# 重新安装
npm install electron --save-dev
{% endraw %}
```

#### 方案2：使用 yarn
```bash
{% raw %}

npm install -g yarn

# 设置镜像
yarn config set electron_mirror https://npmmirror.com/mirrors/electron/

# 安装 Electron
yarn add electron --dev
{% endraw %}
```

#### 方案3：手动下载
```bash
{% raw %}
# 设置环境变量
export ELECTRON_SKIP_BINARY_DOWNLOAD=1

# 安装
npm install electron --save-dev

# 手动下载 Electron
npx electron --version
{% endraw %}
```

#### 方案4：检查 Node.js 版本
```bash
{% raw %}
# 检查 Node.js 版本
node --version

# 推荐版本：18.x 或更高
# 如果版本过低，升级 Node.js
```

## IPC通信问题

### 1. IPC 通信延迟

**问题描述**：
- IPC 调用响应慢
- 大数据传输卡顿

**解决方案**：

#### 方案1：使用流式传输
```javascript
{% raw %}
// 主进程
const fs = require('fs');
const { Readable } = require('stream');

ipcMain.handle('stream-file', async (event, filePath, chunkSize = 64 * 1024) => {
  return new Promise((resolve, reject) => {
    const stream = fs.createReadStream(filePath, { highWaterMark: chunkSize });
    const chunks = [];

    stream.on('data', (chunk) => {
      chunks.push(chunk);
      event.sender.send('stream-progress', {
        received: chunks.length * chunkSize,
        total: fs.statSync(filePath).size,
      });
    });

    stream.on('end', () => {
      const buffer = Buffer.concat(chunks);
      resolve(buffer.toString('base64'));
    });

    stream.on('error', reject);
  });
});
{% endraw %}
```

#### 方案2：批量处理
```javascript
{% raw %}
// 主进程
ipcMain.handle('batch-request', async (event, requests) => {
  const results = [];
  const batchSize = 10;

  for (let i = 0; i < requests.length; i += batchSize) {
    const batch = requests.slice(i, i + batchSize);
    const batchResults = await Promise.all(
      batch.map(req => processRequest(req))
    );
    results.push(...batchResults);
  }

  return results;
});
{% endraw %}
```

#### 方案3：使用 Worker Threads
```javascript
{% raw %}

parentPort.on('message', (message) => {
  const result = heavyComputation(message.data);
  parentPort.postMessage({ type: 'result', data: result });

// 主进程
const { Worker } = require('worker_threads');

ipcMain.handle('compute-heavy', async (event, data) => {
  return new Promise((resolve, reject) => {
    const worker = new Worker('./worker.js');
    worker.on('message', (message) => {
      if (message.type === 'result') {
        resolve(message.data);
        worker.terminate();
      }

    worker.on('error', reject);
    worker.postMessage({ data });
```

```javascript
    // 验证通道
    const validChannels = ['get-data', 'save-data'];
    if (!validChannels.includes(channel)) {
      throw new Error(`Invalid channel: ${channel}`);
    }

      throw new Error('Invalid data format');
```

```javascript
    return false;
  }

  return true;
}
```

#### 方案3：使用对象池

```javascript
class ObjectPool {
  constructor(createFn, size = 10) {
    this.pool = [];
    this.inUse = new Set();
    this.createFn = createFn;

    // 预创建对象
    for (let i = 0; i < size; i++) {
      this.pool.push(createFn());
    }
  }

  acquire() {
    if (this.pool.length > 0) {
      const obj = this.pool.pop();
      this.inUse.add(obj);
      return obj;
    }
    return this.createFn();
  }

  release(obj) {
    if (this.inUse.has(obj)) {
      this.inUse.delete(obj);
      if (obj.reset) {
        obj.reset();
      }
      this.pool.push(obj);
    }
  }
}
```

### 3. 渲染性能差

**问题描述**：
- 页面滚动卡顿
- 动画不流畅
- UI 响应慢

**解决方案**：

#### 方案1：使用虚拟列表
```javascript
{% raw %}
// VirtualList.js
class VirtualList {
  constructor(container, itemHeight, totalItems, renderItem) {
    this.container = container;
    this.itemHeight = itemHeight;
    this.totalItems = totalItems;
    this.renderItem = renderItem;
    this.visibleCount = Math.ceil(container.clientHeight / itemHeight);
    this.startIndex = 0;

    this.init();
  }

  init() {
    this.container.style.height = `${this.totalItems * this.itemHeight}px`;
    this.container.style.position = 'relative';
    this.container.style.overflow = 'auto';

    this.container.addEventListener('scroll', () => {
      this.update();
    });

    this.update();
  }

  update() {
    const scrollTop = this.container.scrollTop;
    const newStartIndex = Math.floor(scrollTop / this.itemHeight);

    if (newStartIndex !== this.startIndex) {
      this.startIndex = newStartIndex;
      this.render();
    }
  }

  render() {
    this.container.innerHTML = '';

    const endIndex = Math.min(
      this.startIndex + this.visibleCount + 2,
      this.totalItems
    );
```

```javascript
function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);

function throttle(func, limit) {
  let inThrottle;
  return function(...args) {
    if (!inThrottle) {
      func.apply(this, args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);

  };
```

```javascript
// utils/pathValidator.js
export function validateFilePath(filePath, options = {}) {
  const { allowedExtensions = [], restrictToDirectory = false } = options;

  if (typeof filePath !== 'string') {
    throw new Error('File path must be a string');
  }

  // 防止路径遍历
  const normalized = path.normalize(filePath);
  if (normalized.includes('..') || normalized.includes('//') || normalized.includes('\\\\')) {
    throw new Error('Path traversal detected');
  }

  // 验证扩展名
  if (allowedExtensions.length > 0) {
    const ext = path.extname(normalized).toLowerCase();
    if (!allowedExtensions.includes(ext)) {
      throw new Error(`File extension .${ext} is not allowed`);
    }
  }

  // 限制为特定目录
  if (restrictToDirectory && !normalized.endsWith('/')) {
    throw new Error('Path must be a directory');
  }

  return normalized;
}
```

```javascript
export function sanitizeHTML(input) {
  if (typeof input !== 'string') return '';

  // 移除危险标签
  const dangerousTags = ['script', 'iframe', 'object', 'embed', 'link'];
  let sanitized = input;

  dangerousTags.forEach(tag => {
    const regex = new RegExp(`<${tag}[^>]*>.*?</${tag}>`, 'gi');
    sanitized = sanitized.replace(regex, '');
  });

  // 移除危险属性
  const dangerousAttrs = ['onload', 'onclick', 'onerror', 'onmouseover'];
  dangerousAttrs.forEach(attr => {
    const regex = new RegExp(`${attr}\\s*=\\s*["'][^"']*["']`, 'gi');
    sanitized = sanitized.replace(regex, '');
  });

  return sanitized;
}

export function sanitizeText(input) {
  if (typeof input !== 'string') return '';

  // 转义 HTML 实体
  return input
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
```

## 总结

Electron 常见问题涵盖了从安装到部署的各个方面。通过了解这些问题及其解决方案，你可以更有效地开发和维护 Electron 应用。

### 关键要点

1. **安装和配置** - 使用镜像源、清理缓存、检查版本
2. **主进程问题** - 窗口管理、内存泄漏、IPC 通信
3. **渲染进程问题** - React 渲染、CSS 样式、热重载
4. **IPC 通信** - 延迟、失败、安全
5. **性能问题** - 启动慢、内存泄漏、渲染性能
6. **安全问题** - 上下文隔离、路径遍历、XSS
7. **打包部署** - 打包失败、自动更新、体积过大
8. **跨平台** - 平台差异、文件路径、系统菜单
9. **调试诊断** - 主进程调试、渲染进程调试、性能分析

### 预防措施

1. **代码审查** - 定期审查代码，发现潜在问题
2. **测试覆盖** - 编写充分的单元测试和集成测试
3. **性能监控** - 定期监控应用性能
4. **安全审计** - 定期进行安全审计
5. **文档更新** - 保持文档与代码同步

### 工具推荐

- **调试工具**: Chrome DevTools, VS Code Debugger
- **性能分析**: Node.js Inspector, Chrome Performance
- **错误监控**: Sentry, Winston
- **测试工具**: Jest, Playwright, Cypress
- **打包工具**: Electron Builder, Electron Packager
