```javascript
// 实现虚拟滚动
class VirtualList {
  constructor(container, itemHeight, totalItems, renderItem) {
    this.container = container;
    this.itemHeight = itemHeight;
    this.totalItems = totalItems;
    this.renderItem = renderItem;
    this.visibleCount = Math.ceil(container.clientHeight / itemHeight);
    this.startIndex = 0;

    this.init();
  }

  init() {
    this.container.style.height = `${this.totalItems * this.itemHeight}px`;
    this.container.style.position = 'relative';
    this.container.style.overflow = 'auto';

    this.container.addEventListener('scroll', () => {
      this.update();
    });

    this.update();
  }

  update() {
    const scrollTop = this.container.scrollTop;
    const newStartIndex = Math.floor(scrollTop / this.itemHeight);

    if (newStartIndex !== this.startIndex) {
      this.startIndex = newStartIndex;
      this.render();
    }
  }

  render() {
    // 清除现有内容
    this.container.innerHTML = '';

    // 只渲染可见项
    const endIndex = Math.min(
      this.startIndex + this.visibleCount + 2,
      this.totalItems
    );

    for (let i = this.startIndex; i < endIndex; i++) {
      const item = this.renderItem(i);
      item.style.position = 'absolute';
      item.style.top = `${i * this.itemHeight}px`;
      item.style.height = `${this.itemHeight}px`;
      item.style.width = '100%';
      this.container.appendChild(item);
    }
  }
}
```

```javascript
// 防抖函数
function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}

// 节流函数
function throttle(func, limit) {
  let inThrottle;
  return function(...args) {
    if (!inThrottle) {
      func.apply(this, args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
}

// 使用示例
```

```javascript

// 输入验证工具
class InputValidator {
  // 验证字符串
  static validateString(input, options = {}) {
    const {
      minLength = 1,
      maxLength = 1000,
      allowedChars = null,
      blacklist = [],
    } = options;

    if (typeof input !== 'string') {
      throw new Error('Input must be a string');
    }

    if (input.length < minLength || input.length > maxLength) {
      throw new Error(`String length must be between ${minLength} and ${maxLength}`);
    }

    if (allowedChars && !new RegExp(`^[${allowedChars}]+$`).test(input)) {
      throw new Error('String contains invalid characters');
    }

    blacklist.forEach(pattern => {
      if (input.includes(pattern)) {
        throw new Error(`String contains blacklisted pattern: ${pattern}`);
      }
    });

    return input;
  }

  // 验证数字
  static validateNumber(input, options = {}) {
    const { min = -Infinity, max = Infinity, integer = false } = options;

    const num = Number(input);
    if (isNaN(num)) {
      throw new Error('Input must be a number');
    }

    if (integer && !Number.isInteger(num)) {
      throw new Error('Input must be an integer');
    }

    if (num < min || num > max) {
      throw new Error(`Number must be between ${min} and ${max}`);
    }

    return num;
  }

  // 验证路径
  static validatePath(input, options = {}) {
    const { allowedExtensions = [], restrictToDirectory = false } = options;

    const path = this.validateString(input, { minLength: 1 });

    // 防止路径遍历攻击
    if (path.includes('..') || path.includes('//') || path.includes('\\\\')) {
      throw new Error('Path traversal detected');
    }

    // 验证文件扩展名
    if (allowedExtensions.length > 0) {
      const ext = path.split('.').pop().toLowerCase();
      if (!allowedExtensions.includes(ext)) {
        throw new Error(`File extension .${ext} is not allowed`);
      }
    }

    // 限制为目录
    if (restrictToDirectory && !path.endsWith('/')) {
      throw new Error('Path must be a directory');
    }

    return path;
  }

  // 验证 URL
  static validateURL(input, options = {}) {
    const { allowedProtocols = ['http:', 'https:'], restrictToDomain = null } = options;

    try {
      const url = new URL(input);

      if (!allowedProtocols.includes(url.protocol)) {
        throw new Error(`Protocol ${url.protocol} is not allowed`);
      }

      if (restrictToDomain && url.hostname !== restrictToDomain) {
        throw new Error(`Domain ${url.hostname} is not allowed`);
      }

      return url.href;
    } catch (error) {
      throw new Error('Invalid URL format');
    }
  }
}

// 使用示例

    min: 0,
    max: 100,
    integer: true,
  });
```

```javascript
// 数据验证函数
function isValidData(data) {
  if (data === null || data === undefined) return false;

  // 防止原型污染
  if (data.__proto__ || data.constructor || data.prototype) {
    return false;
  }

  // 验证数据类型
  if (typeof data === 'object') {
    // 限制对象深度
    if (getObjectDepth(data) > 10) {
      return false;
    }

    // 限制对象大小
    if (JSON.stringify(data).length > 10000) {
      return false;
    }
  }

  return true;
}

function getObjectDepth(obj, depth = 0) {
  if (typeof obj !== 'object' || obj === null) return depth;

  const maxDepth = 10;
  if (depth > maxDepth) return depth;

  return Math.max(...Object.values(obj).map(val =>
    getObjectDepth(val, depth + 1)
  ));
}
```

```javascript

const { dialog } = require('electron');
const fs = require('fs').promises;
const path = require('path');

class SecureFileSystem {
  // 安全的文件读取
  static async readFile(filePath, options = {}) {
    // 验证路径
    const safePath = this.validateFilePath(filePath);

    // 验证文件扩展名
    const allowedExtensions = options.allowedExtensions || ['.txt', '.json', '.md'];
    const ext = path.extname(safePath).toLowerCase();

    if (!allowedExtensions.includes(ext)) {
      throw new Error(`File extension ${ext} is not allowed`);
    }

    // 限制文件大小
    const stats = await fs.stat(safePath);
    const maxSize = options.maxSize || 10 * 1024 * 1024; // 10MB

    if (stats.size > maxSize) {
      throw new Error(`File size ${stats.size} exceeds limit ${maxSize}`);
    }

    // 读取文件
    return fs.readFile(safePath, options.encoding || 'utf8');
  }

  // 安全的文件写入
  static async writeFile(filePath, data, options = {}) {
    // 验证路径
    const safePath = this.validateFilePath(filePath);

    // 验证数据大小
    const maxSize = options.maxSize || 10 * 1024 * 1024; // 10MB
    const dataSize = Buffer.byteLength(data, 'utf8');

    if (dataSize > maxSize) {
      throw new Error(`Data size ${dataSize} exceeds limit ${maxSize}`);
    }

    // 验证数据内容
    if (options.validateContent) {
      this.validateFileContent(data, path.extname(safePath));
    }

    // 创建备份
    if (options.createBackup && await this.fileExists(safePath)) {
      const backupPath = `${safePath}.backup`;
      await fs.copyFile(safePath, backupPath);
    }

    // 写入文件
    return fs.writeFile(safePath, data, options.encoding || 'utf8');
  }

  // 安全的文件选择
  static async selectFile(options = {}) {
    const result = await dialog.showOpenDialog({
      title: options.title || '选择文件',
      filters: options.filters || [{ name: 'All Files', extensions: ['*'] }],
      properties: ['openFile', 'dontAddToRecent'],
    });

    if (result.canceled || result.filePaths.length === 0) {
      return null;
    }

    const filePath = result.filePaths[0];

    // 验证选择的文件
    const safePath = this.validateFilePath(filePath);

    // 验证文件扩展名
    if (options.allowedExtensions) {
      const ext = path.extname(safePath).toLowerCase();
      if (!options.allowedExtensions.includes(ext)) {
        throw new Error(`File extension ${ext} is not allowed`);
      }
    }

    return safePath;
  }

  // 路径验证
  static validateFilePath(filePath) {
    if (typeof filePath !== 'string') {
      throw new Error('File path must be a string');
    }

    // 防止路径遍历
    const normalized = path.normalize(filePath);
    if (normalized.includes('..') || normalized.includes('//') || normalized.includes('\\\\')) {
      throw new Error('Path traversal detected');
    }

    // 限制为特定目录（可选）
    const allowedDirs = options.allowedDirectories || [];
    if (allowedDirs.length > 0) {
      const isInAllowedDir = allowedDirs.some(dir =>
        normalized.startsWith(path.resolve(dir))
      );
      if (!isInAllowedDir) {
        throw new Error('File is not in allowed directory');
      }
    }

    return normalized;
  }

  // 文件内容验证
  static validateFileContent(content, ext) {
    // 根据文件类型验证内容
    switch (ext) {
      case '.json':
        try {
          JSON.parse(content);
        } catch (error) {
          throw new Error('Invalid JSON content');
        }
        break;

      case '.xml':
        if (!content.trim().startsWith('<?xml')) {
          throw new Error('Invalid XML content');
        }
        break;

      case '.html':
      case '.htm':
        // 检查潜在的危险内容
        const dangerousPatterns = [
          /<script[^>]*>[\s\S]*?<\/script>/gi,
          /javascript:/gi,
          /on\w+\s*=/gi,
        ];

        dangerousPatterns.forEach(pattern => {
          if (pattern.test(content)) {
            throw new Error('Potentially dangerous content detected');
          }
        });
        break;
    }
  }

  // 检查文件是否存在
  static async fileExists(filePath) {
    try {
      await fs.access(filePath);
      return true;
    } catch {
      return false;
    }
  }
}

// 使用示例
```

```javascript
// 渲染进程
async function requestWithTimeout(data, timeout = 5000) {
  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), timeout);

  try {
    const result = await window.electronAPI.invoke('request-with-timeout', data);
    clearTimeout(timeoutId);
    return result;
  } catch (error) {
    clearTimeout(timeoutId);
```

```javascript
class WorkerPool {
  constructor(workerPath, poolSize = 4) {
    this.workerPath = workerPath;
    this.poolSize = poolSize;
    this.workers = [];
    this.queue = [];
    this.activeTasks = new Map();

    this.init();
  }

  init() {
    for (let i = 0; i < this.poolSize; i++) {
      const worker = new Worker(this.workerPath);
      this.workers.push(worker);

      worker.on('message', (message) => {
        if (message.type === 'result') {
          const task = this.activeTasks.get(worker);
          if (task) {
            task.resolve(message.data);
            this.activeTasks.delete(worker);
            this.processQueue();
          }
        } else if (message.type === 'error') {
          const task = this.activeTasks.get(worker);
          if (task) {
            task.reject(new Error(message.error));
            this.activeTasks.delete(worker);
            this.processQueue();
          }
        }
      });

      worker.on('error', (error) => {
        console.error('Worker error:', error);
      });
    }
  }

  async execute(data) {
    return new Promise((resolve, reject) => {
      this.queue.push({ data, resolve, reject });
      this.processQueue();
    });
  }

  processQueue() {
    if (this.queue.length === 0) return;

    const availableWorker = this.workers.find(worker =>
      !this.activeTasks.has(worker)
    );

    if (availableWorker) {
      const task = this.queue.shift();
      this.activeTasks.set(availableWorker, task);
      availableWorker.postMessage({ type: 'compute', data: task.data });
    }
  }

  terminate() {
    this.workers.forEach(worker => worker.terminate());
  }
}
```

```javascript
class WindowCommunicator {
  constructor(windowFactory) {
    this.windowFactory = windowFactory;
    this.messageQueues = new Map();
  }

  // 发送消息到指定窗口
  sendToWindow(targetWindowId, channel, data) {
    const targetWindow = this.windowFactory.getWindow(targetWindowId);
    if (!targetWindow) {
      // 如果窗口不存在，加入队列
      if (!this.messageQueues.has(targetWindowId)) {
        this.messageQueues.set(targetWindowId, []);
      }
      this.messageQueues.get(targetWindowId).push({ channel, data });
      return false;
    }

    targetWindow.webContents.send(channel, data);
    return true;
  }

  // 广播消息到所有窗口
  broadcast(channel, data, excludeWindowId = null) {
    const windows = this.windowFactory.getAllWindows();

    windows.forEach(({ id, window }) => {
      if (id !== excludeWindowId) {
        window.webContents.send(channel, data);
      }
    });
  }

  // 发送消息并等待响应
  async sendWithResponse(targetWindowId, channel, data, timeout = 5000) {
    return new Promise((resolve, reject) => {
      const responseChannel = `${channel}-response-${Date.now()}`;
      const timeoutId = setTimeout(() => {
        reject(new Error('Response timeout'));
      }, timeout);

      // 监听响应
      const listener = (event, responseData) => {
        clearTimeout(timeoutId);
        resolve(responseData);
      };

      const targetWindow = this.windowFactory.getWindow(targetWindowId);
      if (!targetWindow) {
        clearTimeout(timeoutId);
        reject(new Error('Target window not found'));
        return;
      }

      targetWindow.webContents.once(responseChannel, listener);

      // 发送消息
      this.sendToWindow(targetWindowId, channel, {
        ...data,
        responseChannel,
      });
    });
  }

  // 处理待处理消息
  processPendingMessages(windowId) {
    const queue = this.messageQueues.get(windowId);
    if (queue) {
      queue.forEach(({ channel, data }) => {
        this.sendToWindow(windowId, channel, data);
      });
      this.messageQueues.delete(windowId);
    }
  }
}
```

```javascript
        };
      }
    });

    return diff;
  }

  // 检查内存泄漏
  checkForLeaks() {
    const usage = this.getMemoryUsage();
    const heap = this.getHeapStatistics();

    const issues = [];

    // 检查堆使用
    if (usage.heapUsed > this.thresholds.heapUsed) {
      issues.push({
        type: 'high-heap-usage',
        severity: 'warning',
        message: `High heap usage: ${(usage.heapUsed / 1024 / 1024).toFixed(2)}MB`,
        value: usage.heapUsed,
        threshold: this.thresholds.heapUsed,
      });
    }

    // 检查堆总大小
    if (usage.heapTotal > this.thresholds.heapTotal) {
      issues.push({
        type: 'high-heap-total',
        severity: 'warning',
        message: `High heap total: ${(usage.heapTotal / 1024 / 1024).toFixed(2)}MB`,
        value: usage.heapTotal,
        threshold: this.thresholds.heapTotal,
      });
    }

    // 检查外部内存
    if (usage.external > this.thresholds.external) {
      issues.push({
        type: 'high-external-memory',
        severity: 'warning',
        message: `High external memory: ${(usage.external / 1024 / 1024).toFixed(2)}MB`,
        value: usage.external,
        threshold: this.thresholds.external,
      });
    }

    // 检查堆碎片
    const heapUsedPercent = (usage.heapUsed / usage.heapTotal) * 100;
    if (heapUsedPercent > 90) {
      issues.push({
        type: 'high-heap-fragmentation',
        severity: 'warning',
        message: `High heap fragmentation: ${heapUsedPercent.toFixed(2)}%`,
        value: heapUsedPercent,
        threshold: 90,
      });
    }

    return issues;
  }

  // 定期监控
  startMonitoring(interval = 30000) {
    this.monitorInterval = setInterval(() => {
      const issues = this.checkForLeaks();

      if (issues.length > 0) {
        console.warn('Memory issues detected:', issues);

        // 触发垃圾回收（如果可用）
        if (global.gc) {
          console.log('Triggering garbage collection...');
          global.gc();
        }
      }

      // 定期创建快照
      if (this.snapshots.length >= 2) {
        const recent = this.snapshots[this.snapshots.length - 1];
        const previous = this.snapshots[this.snapshots.length - 2];
        const diff = this.compareSnapshots(previous, recent);

        // 检查显著增长
        Object.entries(diff.memory).forEach(([key, data]) => {
          if (data.changePercent > 10) { // 增长超过10%
            console.warn(`Significant memory growth in ${key}: ${data.changePercent}`);
          }
        });
      }

      this.takeSnapshot('periodic');
    }, interval);
  }

      this.monitorInterval = null;
    }
  }

    const previous = this.snapshots[this.snapshots.length - 2];

    const report = {
      timestamp: Date.now(),
      latestSnapshot: latest,
      previousSnapshot: previous,
      diff: previous ? this.compareSnapshots(previous, latest) : null,
      issues: this.checkForLeaks(),
```

```javascript
class GarbageCollector {
  constructor() {
    this.gcStrategies = {
      aggressive: () => this.aggressiveGC(),
      moderate: () => this.moderateGC(),
      conservative: () => this.conservativeGC(),
    };
  }

  // 强制垃圾回收
  aggressiveGC() {
    if (global.gc) {
      // 多次调用以确保清理
      for (let i = 0; i < 3; i++) {
        global.gc();
      }
    }
  }

  // 温和垃圾回收
  moderateGC() {
    if (global.gc) {
      global.gc();
    }
  }

  // 保守垃圾回收
  conservativeGC() {
    // 不强制GC，依赖V8自动管理
    // 但可以触发一些清理操作
    if (global.gc) {
      // 只在内存压力大时调用
      const usage = process.memoryUsage();
      if (usage.heapUsed > usage.heapTotal * 0.8) {
        global.gc();
      }
    }
  }

  // 智能垃圾回收
  smartGC(strategy = 'moderate') {
    const gcStrategy = this.gcStrategies[strategy];
    if (gcStrategy) {
      gcStrategy();
    }
  }

  // 内存压力感知的GC
  pressureAwareGC() {
    const usage = process.memoryUsage();
    const heap = require('v8').getHeapStatistics();

    // 计算压力分数
    const heapUsedPercent = (usage.heapUsed / usage.heapTotal) * 100;
    const externalPercent = (usage.external / (100 * 1024 * 1024)) * 100; // 假设100MB限制

    const pressureScore = (heapUsedPercent * 0.7) + (externalPercent * 0.3);

    if (pressureScore > 80) {
      console.log('High memory pressure, triggering GC...');
      this.aggressiveGC();
    } else if (pressureScore > 60) {
      this.moderateGC();
    } else if (pressureScore > 40) {
      this.conservativeGC();
    }
  }

  // 对象池模式
  createObjectPool(createFn, size = 10) {
    const pool = [];
    const inUse = new Set();

    // 预创建对象
    for (let i = 0; i < size; i++) {
      pool.push(createFn());
    }

    return {
      acquire() {
        if (pool.length > 0) {
          const obj = pool.pop();
          inUse.add(obj);
          return obj;
        }
        // 池为空，创建新对象
        const newObj = createFn();
        inUse.add(newObj);
        return newObj;
      },

      release(obj) {
        if (inUse.has(obj)) {
          inUse.delete(obj);
          // 重置对象状态
          if (obj.reset) {
            obj.reset();
          }
          pool.push(obj);
        }
      },

      getStats() {
        return {
          available: pool.length,
          inUse: inUse.size,
          total: pool.length + inUse.size,
        };
      },

      cleanup() {
        pool.length = 0;
        inUse.clear();
      },
    };
  }
}
```

```javascript
class ResourceManager {
  constructor() {
    this.resources = new Map();
    this.cleanupCallbacks = new Map();
  }

  // 注册资源
  register(resourceId, resource, cleanupFn) {
    this.resources.set(resourceId, {
      resource,
      createdAt: Date.now(),
      lastAccessed: Date.now(),
    });

    if (cleanupFn) {
      this.cleanupCallbacks.set(resourceId, cleanupFn);
    }

    return resourceId;
  }

  // 获取资源
  get(resourceId) {
    const entry = this.resources.get(resourceId);
    if (entry) {
      entry.lastAccessed = Date.now();
      return entry.resource;
    }
    return null;
  }

  // 释放资源
  release(resourceId) {
    const entry = this.resources.get(resourceId);
    if (entry) {
      // 执行清理回调
      const cleanupFn = this.cleanupCallbacks.get(resourceId);
      if (cleanupFn) {
        try {
          cleanupFn(entry.resource);
        } catch (error) {
          console.error(`Cleanup error for ${resourceId}:`, error);
        }
      }

      // 从管理器中移除
      this.resources.delete(resourceId);
      this.cleanupCallbacks.delete(resourceId);

      return true;
    }
    return false;
  }

  // 批量释放
  releaseAll(filterFn = null) {
    const released = [];

    this.resources.forEach((entry, resourceId) => {
      if (!filterFn || filterFn(entry, resourceId)) {
        if (this.release(resourceId)) {
          released.push(resourceId);
        }
      }
    });

    return released;
  }

  // 清理不活跃的资源
  cleanupInactive(maxAge = 300000) { // 5分钟
    const now = Date.now();
    const toRelease = [];

    this.resources.forEach((entry, resourceId) => {
      const age = now - entry.lastAccessed;
      if (age > maxAge) {
        toRelease.push(resourceId);
      }
    });

    toRelease.forEach(resourceId => {
      this.release(resourceId);
    });

    return toRelease;
  }

  // 获取统计信息
  getStats() {
    const stats = {
      total: this.resources.size,
      byAge: [],
      byType: {},
    };

    const now = Date.now();

    this.resources.forEach((entry, resourceId) => {
      const age = now - entry.createdAt;
      stats.byAge.push({
        id: resourceId,
        age,
        lastAccessed: entry.lastAccessed,
      });

      // 按类型统计（如果资源有类型属性）
      if (entry.resource.type) {
        if (!stats.byType[entry.resource.type]) {
          stats.byType[entry.resource.type] = 0;
        }
        stats.byType[entry.resource.type]++;
      }
    });

    // 按年龄排序
    stats.byAge.sort((a, b) => a.age - b.age);

    return stats;
  }

  // 导出资源信息
  exportResources() {
    const exported = {};
    this.resources.forEach((entry, resourceId) => {
      exported[resourceId] = {
        createdAt: entry.createdAt,
        lastAccessed: entry.lastAccessed,
        hasCleanup: this.cleanupCallbacks.has(resourceId),
      };
    });
    return exported;
  }
}
```

```javascript
class ErrorMonitor {

      this.handleError(error, 'uncaughtException');
    });

    // 未处理的 Promise 拒绝
    process.on('unhandledRejection', (reason, promise) => {
      this.handleError(reason, 'unhandledRejection', { promise });
    });

    // 警告
    process.on('warning', (warning) => {
      this.handleWarning(warning);
    });

    // 进程错误
    process.on('error', (error) => {
      this.handleError(error, 'processError');
    });

    const errorInfo = {
      timestamp: Date.now(),
      type,
      message: error.message,
      stack: error.stack,
      code: error.code,
      context,
      process: {
        pid: process.pid,
        platform: process.platform,
        version: process.version,
      },
    };

    this.errors.push(errorInfo);

    // 限制错误数量
    if (this.errors.length > 100) {
      this.errors.shift();
    }

    // 通知所有错误处理器
    this.errorHandlers.forEach(handler => {
      try {
        handler(errorInfo);
      } catch (handlerError) {
        console.error('Error handler failed:', handlerError);
      }
    });

    // 记录到控制台
    console.error(`[${type}] ${error.message}`);
    console.error(error.stack);

    // 根据错误类型决定是否退出
    if (type === 'uncaughtException') {
      // 可以选择退出进程
      // process.exit(1);
    }
  }

    const warningInfo = {
      timestamp: Date.now(),
      type: 'warning',
      name: warning.name,
      message: warning.message,
      stack: warning.stack,
    };

    this.errors.push(warningInfo);

    // 通知警告处理器
    this.errorHandlers.forEach(handler => {
      try {

      } catch (handlerError) {

        this.errorHandlers.splice(index, 1);
      }

      total: this.errors.length,
      byType: {},
      byTime: {},
      recent: this.errors.slice(-10),
    };

    const now = Date.now();
    const oneHourAgo = now - 3600000;
    const oneDayAgo = now - 86400000;

    this.errors.forEach(error => {
      // 按类型统计
      if (!stats.byType[error.type]) {
        stats.byType[error.type] = 0;
      }
      stats.byType[error.type]++;

      // 按时间统计

        if (!stats.byTime['lastHour']) {
          stats.byTime['lastHour'] = 0;
        }
        stats.byTime['lastHour']++;
      }

      if (error.timestamp > oneDayAgo) {
        if (!stats.byTime['lastDay']) {
          stats.byTime['lastDay'] = 0;
        }
        stats.byTime['lastDay']++;
      }

      return (now - error.timestamp) <= maxAge;
    });

    return {
      removed: oldCount - this.errors.length,
      remaining: this.errors.length,
    };
  }
}
```

```javascript
      maxFiles: options.maxFiles || 5,
      maxSize: options.maxSize || 10 * 1024 * 1024, // 10MB
      ...options,
    };

    this.init();
  }

  init() {
    // 确保日志目录存在
    const logDir = path.dirname(this.options.file);
    if (!fs.existsSync(logDir)) {
      fs.mkdirSync(logDir, { recursive: true });
    }

    // 创建 Winston logger
    const transports = [];

    // 文件传输
    transports.push(new winston.transports.File({
      filename: this.options.file,
      level: this.options.level,
      maxsize: this.options.maxSize,
      maxFiles: this.options.maxFiles,
      format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.json()
      ),
    }));

    // 控制台传输
    if (this.options.console) {
      transports.push(new winston.transports.Console({
        level: this.options.level,
        format: winston.format.combine(
          winston.format.colorize(),
          winston.format.timestamp(),
          winston.format.printf(({ timestamp, level, message, ...meta }) => {
            return `${timestamp} [${level}]: ${message} ${
              Object.keys(meta).length ? JSON.stringify(meta) : ''
            }`;
          })
        ),
      }));
    }

    this.logger = winston.createLogger({
      level: this.options.level,
      transports,
      exitOnError: false,
    });
  }

  // 日志方法
  log(level, message, meta = {}) {
    this.logger.log(level, message, meta);
  }

  error(message, meta = {}) {
    this.logger.error(message, meta);
  }

  warn(message, meta = {}) {
    this.logger.warn(message, meta);
  }

  info(message, meta = {}) {
    this.logger.info(message, meta);
  }

  debug(message, meta = {}) {
    this.logger.debug(message, meta);
  }

  // 性能日志
  perf(name, start, end, meta = {}) {
    const duration = end - start;
    this.logger.info(`PERF: ${name}`, {
      duration,
      start,
      end,
      ...meta,
    });
  }

  // 错误日志
  errorWithStack(error, meta = {}) {
    this.logger.error(error.message, {
      stack: error.stack,
      code: error.code,
      ...meta,
    });
  }

  // 获取日志文件大小
  getFileSize() {
    try {
      const stats = fs.statSync(this.options.file);
      return stats.size;
    } catch (error) {
      return 0;
    }
  }

  // 清理旧日志
  cleanupOldLogs(maxAge = 7 * 24 * 60 * 60 * 1000) { // 7天
    const logDir = path.dirname(this.options.file);
    const files = fs.readdirSync(logDir);

    const now = Date.now();
    const removed = [];

    files.forEach(file => {
      if (file.startsWith('app') && file.endsWith('.log')) {
        const filePath = path.join(logDir, file);
        const stats = fs.statSync(filePath);

        if ((now - stats.mtimeMs) > maxAge) {
          fs.unlinkSync(filePath);
          removed.push(file);
        }
      }
    });

    return removed;
  }

  // 获取日志统计
  getStats() {
    const size = this.getFileSize();
    const logDir = path.dirname(this.options.file);
    const files = fs.readdirSync(logDir);

    const logFiles = files
      .filter(file => file.startsWith('app') && file.endsWith('.log'))
      .map(file => {
        const filePath = path.join(logDir, file);
        const stats = fs.statSync(filePath);
        return {
          name: file,
          size: stats.size,
          modified: stats.mtime,
        };
      })
      .sort((a, b) => b.modified - a.modified);

    return {
      currentSize: size,
      totalFiles: logFiles.length,
      files: logFiles,
    };
  }
}
```

```javascript
    const modulePaths = paths[moduleName];
    if (!modulePaths) {

    }

    const platformPaths = modulePaths[this.platform];
    if (!platformPaths) {
      throw new Error(`Native module ${moduleName} not supported on ${this.platform}`);

  async load(moduleName) {
```

```javascript
    const channel = this.channels.get(channelName);
    if (!channel) {
      throw new Error(`Channel ${channelName} not found`);
    }

    // 发送到两个端口
    channel.port1.postMessage(message, transfer);
    channel.port2.postMessage(message, transfer);
  }

  // 接收消息
  receive(channelName, callback, port = 'port1') {
    const channel = this.channels.get(channelName);
    if (!channel) {
      throw new Error(`Channel ${channelName} not found`);
    }

    const messagePort = channel[port];

    messagePort.onmessage = (event) => {
      callback(event.data);
    };

    // 返回清理函数
    return () => {
      messagePort.onmessage = null;
    };
  }

  // 关闭通道
  close(channelName) {
    const channel = this.channels.get(channelName);
    if (channel) {
      channel.port1.close();
      channel.port2.close();
      this.channels.delete(channelName);
      this.messagePorts.delete(channelName);
    }
  }
}
```

```javascript
// 插件管理器
class PluginManager {
  constructor() {
    this.plugins = new Map();
    this.hooks = new Map();
    this.pluginDir = path.join(process.cwd(), 'plugins');
  }

  // 加载插件
  async loadPlugin(pluginName) {
    const pluginPath = path.join(this.pluginDir, pluginName);

    try {
      // 检查插件目录
      if (!fs.existsSync(pluginPath)) {
        throw new Error(`Plugin directory not found: ${pluginPath}`);
      }

      // 加载插件配置
      const configPath = path.join(pluginPath, 'plugin.json');
      if (!fs.existsSync(configPath)) {
        throw new Error(`Plugin config not found: ${configPath}`);
      }

      const config = JSON.parse(fs.readFileSync(configPath, 'utf8'));

      // 验证插件配置
      this.validatePluginConfig(config);

      // 加载插件模块
      const mainPath = path.join(pluginPath, config.main || 'index.js');
      const pluginModule = require(mainPath);

      // 创建插件实例
      const plugin = {
        name: config.name,
        version: config.version,
        description: config.description,
        author: config.author,
        config,
        module: pluginModule,
        enabled: true,
        loadedAt: Date.now(),
      };

      // 初始化插件
      if (pluginModule.init) {
        await pluginModule.init(this);
      }

      // 注册插件
      this.plugins.set(pluginName, plugin);

      console.log(`Plugin loaded: ${pluginName} v${plugin.version}`);
      return plugin;

    } catch (error) {
      console.error(`Failed to load plugin ${pluginName}:`, error);
      throw error;
    }
  }

  // 卸载插件
  async unloadPlugin(pluginName) {
    const plugin = this.plugins.get(pluginName);
    if (!plugin) {
      throw new Error(`Plugin ${pluginName} not found`);
    }

    // 调用卸载钩子
    if (plugin.module.destroy) {
      await plugin.module.destroy(this);
    }

    // 清理钩子
    this.hooks.forEach((hooks, hookName) => {
      hooks.forEach((hook, index) => {
        if (hook.plugin === pluginName) {
          hooks.splice(index, 1);
        }
      });
    });

    // 从缓存中移除
    this.plugins.delete(pluginName);

    // 清除 require 缓存
    const pluginPath = path.join(this.pluginDir, pluginName);
    Object.keys(require.cache).forEach(key => {
      if (key.startsWith(pluginPath)) {
        delete require.cache[key];
      }
    });

    console.log(`Plugin unloaded: ${pluginName}`);
  }

  // 注册钩子
  registerHook(hookName, callback, pluginName = 'system') {
    if (!this.hooks.has(hookName)) {
      this.hooks.set(hookName, []);
    }

    const hooks = this.hooks.get(hookName);
    hooks.push({ callback, plugin: pluginName });

    // 返回取消注册函数
    return () => {
      const index = hooks.findIndex(h => h.callback === callback);
      if (index > -1) {
        hooks.splice(index, 1);
      }
    };
  }

  // 触发钩子
  async triggerHook(hookName, ...args) {
    const hooks = this.hooks.get(hookName);
    if (!hooks) {
      return [];
    }

    const results = [];

    for (const hook of hooks) {
      try {
        const result = await hook.callback(...args);
        results.push({ plugin: hook.plugin, result });
      } catch (error) {
        console.error(`Hook ${hookName} error from plugin ${hook.plugin}:`, error);
        results.push({ plugin: hook.plugin, error: error.message });
      }
    }

    return results;
  }

  // 获取插件列表
  getPlugins() {
    return Array.from(this.plugins.entries()).map(([name, plugin]) => ({
      name,
      version: plugin.version,
      description: plugin.description,
      author: plugin.author,
      enabled: plugin.enabled,
      loadedAt: plugin.loadedAt,
    }));
  }

  // 启用/禁用插件
  togglePlugin(pluginName, enabled) {
    const plugin = this.plugins.get(pluginName);
    if (!plugin) {
      throw new Error(`Plugin ${pluginName} not found`);
    }

    plugin.enabled = enabled;

    if (plugin.module.onToggle) {
      plugin.module.onToggle(this, enabled);
    }

    return enabled;
  }

  // 验证插件配置
  validatePluginConfig(config) {
    const required = ['name', 'version', 'main'];

    required.forEach(field => {
      if (!config[field]) {
        throw new Error(`Missing required field: ${field}`);
      }
    });

    // 验证版本格式
    if (!/^\d+\.\d+\.\d+$/.test(config.version)) {
      throw new Error('Invalid version format. Use semver (e.g., 1.0.0)');
    }
  }

  // 扫描插件目录
  async scanPlugins() {
    if (!fs.existsSync(this.pluginDir)) {
      return [];
    }

    const files = fs.readdirSync(this.pluginDir);
    const plugins = [];

    for (const file of files) {
      const pluginPath = path.join(this.pluginDir, file);
      const stat = fs.statSync(pluginPath);

      if (stat.isDirectory()) {
        try {
          const plugin = await this.loadPlugin(file);
          plugins.push(plugin);
        } catch (error) {
          console.warn(`Failed to load plugin ${file}:`, error.message);
        }
      }
    }

    return plugins;
  }
}
```

```javascript
// 热重载管理器
class HotReloadManager {
  constructor(options = {}) {
    this.options = {
      watchDirs: options.watchDirs || ['src', 'plugins'],
      ignorePatterns: options.ignorePatterns || [
        /node_modules/,
        /\.git/,
        /\.DS_Store/,
        /__pycache__/,
      ],
      debounceDelay: options.debounceDelay || 1000,
      ...options,
    };

    this.watchers = new Map();
    this.changeHandlers = new Map();
    this.debounceTimers = new Map();
  }

  // 开始监视目录
  watchDirectory(dirPath, handler) {
    if (this.watchers.has(dirPath)) {
      console.warn(`Already watching ${dirPath}`);
      return;
    }

    const watcher = fs.watch(dirPath, {
      recursive: true,
      persistent: true,
    }, (eventType, filename) => {
      // 检查是否应该忽略
      if (this.shouldIgnore(filename)) {
        return;
      }

      // 防抖处理
      const timerKey = `${dirPath}:${filename}`;

      if (this.debounceTimers.has(timerKey)) {
        clearTimeout(this.debounceTimers.get(timerKey));
      }

      const timer = setTimeout(() => {
        console.log(`File changed: ${filename} (${eventType})`);

        // 触发处理器
        if (handler) {
          handler({ eventType, filename, dirPath });
        }

        // 触发全局处理器
        this.triggerChangeHandlers({ eventType, filename, dirPath });

        this.debounceTimers.delete(timerKey);
      }, this.options.debounceDelay);

      this.debounceTimers.set(timerKey, timer);
    });

    this.watchers.set(dirPath, watcher);
    this.changeHandlers.set(dirPath, handler);

    console.log(`Started watching: ${dirPath}`);
  }

  // 停止监视
  stopWatching(dirPath) {
    const watcher = this.watchers.get(dirPath);
    if (watcher) {
      watcher.close();
      this.watchers.delete(dirPath);
      this.changeHandlers.delete(dirPath);
      console.log(`Stopped watching: ${dirPath}`);
    }
  }

  // 停止所有监视
  stopAll() {
    this.watchers.forEach((watcher, dirPath) => {
      watcher.close();
    });
    this.watchers.clear();
    this.changeHandlers.clear();
    this.debounceTimers.clear();
    console.log('Stopped all watchers');
  }

  // 添加全局变化处理器
  addChangeHandler(handler) {
    const id = Date.now().toString();
    this.changeHandlers.set(id, handler);

    // 返回移除函数
    return () => {
      this.changeHandlers.delete(id);
    };
  }

  // 触发变化处理器
  triggerChangeHandlers(change) {
    this.changeHandlers.forEach((handler, key) => {
      if (typeof handler === 'function' && !key.includes(':')) {
        try {
          handler(change);
        } catch (error) {
          console.error('Change handler error:', error);
        }
      }
    });
  }

  // 检查是否应该忽略
  shouldIgnore(filename) {
    if (!filename) return true;

    return this.options.ignorePatterns.some(pattern => {
      if (typeof pattern === 'string') {
        return filename.includes(pattern);
      } else if (pattern instanceof RegExp) {
        return pattern.test(filename);
      }
      return false;
    });
  }

  // 获取监视状态
  getStatus() {
    return {
      watching: Array.from(this.watchers.keys()),
      totalWatchers: this.watchers.size,
      pendingChanges: this.debounceTimers.size,
    };
  }

  // 重新加载模块
  reloadModule(modulePath) {
    try {
      // 清除 require 缓存
      Object.keys(require.cache).forEach(key => {
        if (key.startsWith(modulePath)) {
          delete require.cache[key];
        }
      });

      // 重新加载模块
      const reloaded = require(modulePath);
      console.log(`Reloaded module: ${modulePath}`);

      return reloaded;
    } catch (error) {
      console.error(`Failed to reload module ${modulePath}:`, error);
      throw error;
    }
  }
}

// 使用示例
const hotReload = new HotReloadManager({
  watchDirs: ['src', 'plugins'],
  ignorePatterns: [/node_modules/, /\.git/, /\.DS_Store/],
  debounceDelay: 1000,
});

// 开始监视
hotReload.watchDirectory('src', (change) => {
  console.log('Source file changed:', change);

  // 重新加载相关模块
  if (change.filename.endsWith('.js')) {
    const modulePath = path.join(process.cwd(), 'src', change.filename);
    try {
      hotReload.reloadModule(modulePath);
    } catch (error) {
      console.error('Reload failed:', error);
    }
  }
});

// 添加全局处理器
const removeHandler = hotReload.addChangeHandler((change) => {
  console.log('Global change detected:', change);
});

// 获取状态
const status = hotReload.getStatus();
console.log('Hot reload status:', status);
```
