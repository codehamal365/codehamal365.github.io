// ✅ 推荐：及时释放资源
// main/utils/resourceManager.ts
export class ResourceManager {
  constructor() {
    this.resources = new Map();
  }

  register(resourceId, resource, cleanupFn) {
    this.resources.set(resourceId, { resource, cleanupFn });
  }

  release(resourceId) {
    const entry = this.resources.get(resourceId);
    if (entry) {
      if (entry.cleanupFn) {
        entry.cleanupFn(entry.resource);
      }
      this.resources.delete(resourceId);
    }
  }

  releaseAll() {
    this.resources.forEach((entry, resourceId) => {
      this.release(resourceId);
    });
  }
}

// main/window/windowManager.ts
import { ResourceManager } from '../utils/resourceManager';

export class WindowManager {
  constructor() {
    this.resourceManager = new ResourceManager();
    this.windows = new Map();
  }

  createWindow(name, config) {
    const window = new BrowserWindow(config);
    const windowId = `${name}-${Date.now()}`;

    // 注册资源
    this.resourceManager.register(windowId, window, (w) => {
      if (!w.isDestroyed()) {
        w.destroy();
      }
    });

    this.windows.set(windowId, window);

    // 窗口关闭时清理
    window.on('closed', () => {
      this.resourceManager.release(windowId);
      this.windows.delete(windowId);
    });

    return windowId;
  }

  cleanup() {
    this.resourceManager.releaseAll();
    this.windows.clear();
  }
}

// main/utils/eventManager.ts
export class EventManager {
  constructor() {
    this.listeners = new Map();
  }

  on(event, callback, context = null) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }

    const listener = { callback, context };
    this.listeners.get(event).push(listener);

    // 返回取消函数
    return () => {
      this.off(event, callback);
    };
  }

  off(event, callback) {
    const listeners = this.listeners.get(event);
    if (listeners) {
      const index = listeners.findIndex(l => l.callback === callback);
      if (index > -1) {
        listeners.splice(index, 1);
      }
    }
  }

  emit(event, ...args) {
    const listeners = this.listeners.get(event);
    if (listeners) {
      listeners.forEach(({ callback, context }) => {
        try {
          callback.call(context, ...args);
        } catch (error) {
          console.error(`Error in event listener for ${event}:`, error);
        }
      });
    }
  }

  removeAllListeners(event) {
    if (event) {
      this.listeners.delete(event);
    } else {
      this.listeners.clear();
    }
  }
}

// ✅ 推荐：使用缓存策略
// renderer/utils/cache.ts
export class Cache {
  constructor(options = {}) {
    this.ttl = options.ttl || 60000; // 1分钟
    this.maxSize = options.maxSize || 100;
    this.store = new Map();
  }

  set(key, value) {
    const item = {
      value,
      timestamp: Date.now(),
    };

    // 检查容量
    if (this.store.size >= this.maxSize) {
      // 移除最旧的项
      const oldestKey = this.getOldestKey();
      if (oldestKey) {
        this.store.delete(oldestKey);
      }
    }

    this.store.set(key, item);
  }

  get(key) {
    const item = this.store.get(key);
    if (!item) return null;

    // 检查过期
    if (Date.now() - item.timestamp > this.ttl) {
      this.store.delete(key);
      return null;
    }

    return item.value;
  }

  has(key) {
    return this.get(key) !== null;
  }

  clear() {
    this.store.clear();
  }

  getOldestKey() {
    let oldestKey = null;
    let oldestTime = Infinity;

    this.store.forEach((item, key) => {
      if (item.timestamp < oldestTime) {
        oldestTime = item.timestamp;
        oldestKey = key;
      }
    });

    return oldestKey;
  }
}

// renderer/utils/api.ts
import { Cache } from './cache';

const apiCache = new Cache({ ttl: 30000 }); // 30秒

export const fetchData = async (url) => {
  // 检查缓存
  const cached = apiCache.get(url);
  if (cached) {
    return cached;
  }

  // 发起请求
  const response = await fetch(url);
  const data = await response.json();

  // 缓存结果
  apiCache.set(url, data);

  return data;
};

// ✅ 推荐：严格的输入验证
// main/utils/validator.ts
export class Validator {
  // 验证字符串
  static validateString(input, options = {}) {
    const {
      minLength = 1,
      maxLength = 1000,
      allowedChars = null,
      blacklist = [],
    } = options;

    if (typeof input !== 'string') {
      throw new Error('Input must be a string');
    }

    if (input.length < minLength || input.length > maxLength) {
      throw new Error(`String length must be between ${minLength} and ${maxLength}`);
    }

    if (allowedChars && !new RegExp(`^[${allowedChars}]+$`).test(input)) {
      throw new Error('String contains invalid characters');
    }

    blacklist.forEach(pattern => {
      if (input.includes(pattern)) {
        throw new Error(`String contains blacklisted pattern: ${pattern}`);
      }
    });

    return input;
  }

  // 验证路径
  static validatePath(input, options = {}) {
    const { allowedExtensions = [], restrictToDirectory = false } = options;

    const path = this.validateString(input, { minLength: 1 });

    // 防止路径遍历
    if (path.includes('..') || path.includes('//') || path.includes('\\\\')) {
      throw new Error('Path traversal detected');
    }

    // 验证扩展名
    if (allowedExtensions.length > 0) {
      const ext = path.split('.').pop().toLowerCase();
      if (!allowedExtensions.includes(ext)) {
        throw new Error(`File extension .${ext} is not allowed`);
      }
    }

    return path;
  }

  // 验证数字
  static validateNumber(input, options = {}) {
    const { min = -Infinity, max = Infinity, integer = false } = options;

    const num = Number(input);
    if (isNaN(num)) {
      throw new Error('Input must be a number');
    }

    if (integer && !Number.isInteger(num)) {
      throw new Error('Input must be an integer');
    }

    if (num < min || num > max) {
      throw new Error(`Number must be between ${min} and ${max}`);
    }

    return num;
  }
}

// main/utils/fileSystem.ts
import fs from 'fs/promises';
import path from 'path';

export class SecureFileSystem {
  // 安全的文件读取
  static async readFile(filePath, options = {}) {
    // 验证路径
    const safePath = this.validateFilePath(filePath);

    // 验证文件扩展名
    const allowedExtensions = options.allowedExtensions || ['.txt', '.json', '.md'];
    const ext = path.extname(safePath).toLowerCase();

    if (!allowedExtensions.includes(ext)) {
      throw new Error(`File extension ${ext} is not allowed`);
    }

    // 限制文件大小
    const stats = await fs.stat(safePath);
    const maxSize = options.maxSize || 10 * 1024 * 1024; // 10MB

    if (stats.size > maxSize) {
      throw new Error(`File size ${stats.size} exceeds limit ${maxSize}`);
    }

    // 读取文件
    return fs.readFile(safePath, options.encoding || 'utf8');
  }

  // 安全的文件写入
  static async writeFile(filePath, data, options = {}) {
    // 验证路径
    const safePath = this.validateFilePath(filePath);

    // 验证数据大小
    const maxSize = options.maxSize || 10 * 1024 * 1024; // 10MB
    const dataSize = Buffer.byteLength(data, 'utf8');

    if (dataSize > maxSize) {
      throw new Error(`Data size ${dataSize} exceeds limit ${maxSize}`);
    }

    // 创建备份
    if (options.createBackup && await this.fileExists(safePath)) {
      const backupPath = `${safePath}.backup`;
      await fs.copyFile(safePath, backupPath);
    }

    // 写入文件
    return fs.writeFile(safePath, data, options.encoding || 'utf8');
  }

  // 路径验证
  static validateFilePath(filePath) {
    if (typeof filePath !== 'string') {
      throw new Error('File path must be a string');
    }

    // 防止路径遍历
    const normalized = path.normalize(filePath);
    if (normalized.includes('..') || normalized.includes('//') || normalized.includes('\\\\')) {
      throw new Error('Path traversal detected');
    }

    return normalized;
  }

  // 检查文件是否存在
  static async fileExists(filePath) {
    try {
      await fs.access(filePath);
      return true;
    } catch {
      return false;
    }
  }
}

      this.isChecking = false;

// main/services/performanceService.js
export class PerformanceService {
  constructor() {
    this.metrics = new Map();
    this.startTime = Date.now();
  }

  startMeasurement(name) {
    const start = process.hrtime.bigint();
    this.metrics.set(name, { start, end: null, duration: null });
  }

  endMeasurement(name) {
    const metric = this.metrics.get(name);
    if (metric && metric.start) {
      const end = process.hrtime.bigint();
      metric.end = end;
      metric.duration = Number(end - metric.start) / 1000000; // 转换为毫秒
    }
  }

  getMetrics() {
    const result = {};
    this.metrics.forEach((metric, name) => {
      result[name] = {
        duration: metric.duration,
        start: metric.start ? Number(metric.start) : null,
        end: metric.end ? Number(metric.end) : null,
      };
    });

    result.appUptime = Date.now() - this.startTime;

    return result;
  }

  logMetrics() {
    const metrics = this.getMetrics();
    console.log('Performance Metrics:', metrics);
  }

  // 内存使用监控
  getMemoryUsage() {
    const usage = process.memoryUsage();
    return {
      rss: usage.rss / 1024 / 1024, // MB
      heapTotal: usage.heapTotal / 1024 / 1024,
      heapUsed: usage.heapUsed / 1024 / 1024,
      external: usage.external / 1024 / 1024,
    };
  }

  // CPU 使用监控
  getCPUUsage() {
    const cpuUsage = process.cpuUsage();
    return {
      user: cpuUsage.user / 1000000, // 转换为秒
      system: cpuUsage.system / 1000000,
    };
  }
}

import path from 'path';
import fs from 'fs';

export class LogService {
  constructor(options = {}) {

      file: options.file || path.join(process.cwd(), 'logs', 'app.log'),
      maxFiles: options.maxFiles || 7,
      maxSize: options.maxSize || 10 * 1024 * 1024, // 10MB

    this.init();
  }

  init() {
    // 确保日志目录存在

    if (!fs.existsSync(logDir)) {
      fs.mkdirSync(logDir, { recursive: true });
    }

      level: this.options.level,
      format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.json()
      ),
      transports: [
        new winston.transports.File({
          filename: this.options.file,
          maxsize: this.options.maxSize,
          maxFiles: this.options.maxFiles,
        }),
        new winston.transports.Console({
          format: winston.format.combine(
            winston.format.colorize(),
            winston.format.simple()
          ),
        }),
      ],
    });
  }

  log(level, message, meta = {}) {
    this.logger.log(level, message, meta);
  }

  error(message, meta = {}) {
    this.logger.error(message, meta);
  }

  warn(message, meta = {}) {
    this.logger.warn(message, meta);
  }

  info(message, meta = {}) {
    this.logger.info(message, meta);
  }

  debug(message, meta = {}) {
    this.logger.debug(message, meta);
  }

  // 清理旧日志
  cleanupOldLogs(maxAge = 7 * 24 * 60 * 60 * 1000) { // 7天
    const logDir = path.dirname(this.options.file);
    const files = fs.readdirSync(logDir);

    const now = Date.now();
    const removed = [];

    files.forEach(file => {
      if (file.startsWith('app') && file.endsWith('.log')) {
        const filePath = path.join(logDir, file);
        const stats = fs.statSync(filePath);

        if ((now - stats.mtimeMs) > maxAge) {
          fs.unlinkSync(filePath);
          removed.push(file);
        }
      }
    });

    return removed;
  }

  // 获取日志统计
  getStats() {
    const logDir = path.dirname(this.options.file);
    const files = fs.readdirSync(logDir);

    const logFiles = files
      .filter(file => file.startsWith('app') && file.endsWith('.log'))
      .map(file => {
        const filePath = path.join(logDir, file);
        const stats = fs.statSync(filePath);
        return {
          name: file,
          size: stats.size,
          modified: stats.mtime,
        };
      })
      .sort((a, b) => b.modified - a.modified);

    return {
      totalFiles: logFiles.length,
      files: logFiles,
    };
  }
}

// ✅ 推荐：实现数据备份
// main/services/backupService.js
import fs from 'fs/promises';
import path from 'path';
import archiver from 'archiver';

export class BackupService {
  constructor(backupDir) {
    this.backupDir = backupDir;
  }

  async createBackup(data, name = 'backup') {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupName = `${name}-${timestamp}.zip`;
    const backupPath = path.join(this.backupDir, backupName);

    // 确保备份目录存在
    await fs.mkdir(this.backupDir, { recursive: true });

    // 创建压缩包
    const output = fs.createWriteStream(backupPath);
    const archive = archiver('zip', { zlib: { level: 9 } });

    return new Promise((resolve, reject) => {
      output.on('close', () => {
        resolve({
          path: backupPath,
          size: archive.pointer(),
          name: backupName,
        });
      });

      archive.on('error', reject);

      archive.pipe(output);

      // 添加数据到压缩包
      if (typeof data === 'string') {
        archive.append(data, { name: 'data.json' });
      } else {
        archive.append(JSON.stringify(data, null, 2), { name: 'data.json' });
      }

      archive.finalize();
    });
  }

  async restoreBackup(backupPath) {
    const data = await fs.readFile(backupPath, 'utf8');
    return JSON.parse(data);
  }

  async listBackups() {
    const files = await fs.readdir(this.backupDir);
    const backups = [];

    for (const file of files) {
      if (file.endsWith('.zip')) {
        const filePath = path.join(this.backupDir, file);
        const stats = await fs.stat(filePath);
        backups.push({
          name: file,
          path: filePath,
          size: stats.size,
          created: stats.birthtime,
        });
      }
    }

    return backups.sort((a, b) => b.created - a.created);
  }

  async cleanupOldBackups(maxAge = 30 * 24 * 60 * 60 * 1000) { // 30天
    const backups = await this.listBackups();
    const now = Date.now();
    const removed = [];

    for (const backup of backups) {
      if ((now - backup.created.getTime()) > maxAge) {
        await fs.unlink(backup.path);
        removed.push(backup.name);
      }
    }

    return removed;
  }
}

// main/main.js

const backupService = new BackupService(path.join(process.cwd(), 'backups'));

// 定期备份

      timestamp: Date.now(),

    // 清理旧备份

    if (removed.length > 0) {
      console.log('Cleaned up old backups:', removed);
    }
  } catch (error) {
    console.error('Backup failed:', error);
  }
